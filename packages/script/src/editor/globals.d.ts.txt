/** 脚本虚拟程序专用声明资源，不参与宿主 TypeScript 程序。 */
/// <reference path="../quickjs/runtime/plotKind.ts" />
import type { InputDef, InputOptions } from '@363045841yyt/klinechart-plot-contracts'
import type { LibraryModules } from '../contracts/library.js'
import type { TechnicalAnalysisLibrary } from '../Library/types.js'
import { PaneKind as PaneKindEnum } from '../quickjs/types.js'
import type { ScriptParamValue } from '../quickjs/types.js'
import type { Series } from '../series/types.js'
import type { ScriptIndicatorDeclaration, ScriptPlotOptions } from './types.js'

declare global {
  /** 编辑器从当前 inputs 推导并合并具体参数。 */
  interface ScriptParameters {}

  const open: Series
  const high: Series
  const low: Series
  const close: Series
  const volume: Series
  const time: Series
  /**
   * 内建技术分析库
   * 序列结果用 [0] 取当前值、[1] 取上一根值，例如 ta.sma(close, 20)[0]。
   */
  const ta: TechnicalAnalysisLibrary
  const params: ScriptParameters

  /** 指标注册位置枚举；取值与宿主 PaneKind 一致，由宿主注入 VM。 */
  const PaneKind: typeof PaneKindEnum

  /** 声明输入默认值与展示选项；options 只在字符串参数上生成下拉。 */
  function input<T extends ScriptParamValue>(value: T, options?: InputOptions): InputDef<T>
  /** 定义一个指标 */
  function defineIndicator(definition: ScriptIndicatorDeclaration): void
  /**
   * 在当前 K 线上绘制形状或字符。
   * @param kind 只能是 PlotKind.Shape 或 PlotKind.Char。
   * @param value 标记信号；absolute 定位时为价格。null 或 false 不绘制。
   * @param options 必填 id；可设置颜色、位置、大小和文字。
   */
  function plot(kind: PlotKind.Shape | PlotKind.Char, value: number | boolean | null, options: ScriptPlotOptions): void
  /**
   * 在当前 K 线位置上绘制指定图形。
   * @param kind 绘图类型：
   *
   * | 类型 | 作用 |
   * | --- | --- |
   * | PlotKind.Line | 折线 |
   * | PlotKind.Histogram | 柱状图 |
   * | PlotKind.Points | 散点 |
   * | PlotKind.Shape | 形状标记 |
   * | PlotKind.Char | 字符标记 |
   * | PlotKind.Arrow | 数值箭头，正负决定方向，绝对值决定长度 |
   *
   * @param value 当前 K 线的绘图值；null 不绘制。
   * @param options 必填 id；其余为对应图形的样式选项。
   */
  function plot(kind: PlotKind, value: number | null, options: ScriptPlotOptions): void
  /** 注册一个库，其成员同时绑定到同名全局对象。 */
  function defineLibrary<T extends object>(name: string, members: T): Readonly<T>
  /** 读取已注册的库，内置名称由统一库类型目录提供。 */
  function importLibrary<K extends keyof LibraryModules>(name: K): LibraryModules[K]
}
